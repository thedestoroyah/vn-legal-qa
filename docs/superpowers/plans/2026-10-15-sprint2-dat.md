```python
def _strip(text):
    return text.replace("*", "").strip()


def _article_lines(articles):
    for a in articles:
        yield f"Điều {a['number']}. {_strip(a['title'])}"
        if a["text"]:
            yield _strip(a["text"])
        for c in a["clauses"]:
            yield f"{c['number']}. {_strip(c['text'])}"
            for p in c["points"]:
                yield f"{p['number']}) {_strip(p['text'])}"


def _chapter_lines(chapters):
    for ch in chapters:
        yield f"Chương {ch['number']}"
        yield _strip(ch["title"])
        yield from _article_lines(ch["articles"])
        for sec in ch["sections"]:
            yield f"Mục {sec['number']}"
            yield _strip(sec["title"])
            yield from _article_lines(sec["articles"])


def render_text(law):
    """Dựng lại văn bản theo quy ước dòng mà chunk_law/fixed_chunk đọc; bỏ tiêu đề Phần vì không phải đơn vị trích dẫn."""
    def lines():
        yield from _article_lines(law["articles"])
        for part in law["parts"]:
            yield from _article_lines(part["articles"])
            yield from _chapter_lines(part["chapters"])
        yield from _chapter_lines(law["chapters"])
    return "\n".join(lines())


def article_key(chunk_id):
    return _CLAUSE_SUFFIX_RE.sub("", chunk_id)


def article_chunks(chunks):
    """Một chunk mỗi Điều (biến thể RQ2): dẫn nhập rồi các Khoản đánh số, theo thứ tự."""
    out = {}
    for c in chunks:
        key = article_key(c["chunk_id"])
        piece = f"{c['clause'].removeprefix('Khoản ')}. {c['content']}" if c["clause"] else c["content"]
        prev = out.get(key)
        out[key] = ({**prev, "content": f"{prev['content']}\n{piece}"} if prev
                    else {**c, "chunk_id": key, "clause": None, "content": piece})
    return list(out.values())


def build_all(structured_dir, laws):
    """(chunk Khoản, chunk Điều, chunk cố định) cho mọi file JSON có số hiệu khai báo trong laws.json."""
    by_number = {info["number"].upper(): {"law_id": law_id, **info} for law_id, info in laws.items()}
    clause, fixed = [], []
    for path in sorted(Path(structured_dir).glob("*.json")):
        law = json.loads(path.read_text(encoding="utf-8"))
        meta = by_number.get(law["law_number"].upper())
        if meta is None:
            print(f"bỏ qua {path.name}: số hiệu {law['law_number']} chưa có trong laws.json", file=sys.stderr)
            continue
        text = render_text(law)
        part = chunk_law(text, meta)
        clause = clause + part
        fixed = fixed + fixed_chunk(text, meta)
        print(f"{meta['law_id']}: {len({c['article'] for c in part})} điều, {len(part)} chunk")
    return clause, article_chunks(clause), fixed


def read_jsonl(path):
    return [json.loads(line) for line in Path(path).read_text(encoding="utf-8").splitlines() if line.strip()]


def write_jsonl(path, rows):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8") as f:
        for row in rows:
            f.write(json.dumps(row, ensure_ascii=False) + "\n")


def main():
    argparse.ArgumentParser(description=__doc__).parse_args()
    laws = json.loads(LAWS_PATH.read_text(encoding="utf-8"))
    clause, article, fixed = build_all(STRUCTURED_DIR, laws)
    if not clause:
        sys.exit(f"không có văn bản nào trong {STRUCTURED_DIR}/ khớp laws.json")
    for path, rows in ((OUT_PATH, clause), (ARTICLE_OUT_PATH, article), (FIXED_OUT_PATH, fixed)):
        write_jsonl(path, rows)
        print(f"{path}: {len(rows)} chunk")


if __name__ == "__main__":
    main()
```

```python
"""Schema và kiểm tra hợp lệ tập câu hỏi kiểm thử (Phụ lục A.4)."""
from datetime import date

from src.ingestion.legal_chunker import article_key

TYPES = {"situational", "temporal", "direct", "out_of_scope"}
DOMAINS = {"dan_su", "lao_dong", "doanh_nghiep", "thuong_mai", "ngoai_pham_vi"}
SOURCES = {"zalo", "synthetic", "manual"}
FIELDS = ("id", "question", "type", "domain", "event_date", "gold_citations", "reference_answer", "source",
          "checked_by")


def is_iso_date(value):
    try:
        date.fromisoformat(value)
    except (TypeError, ValueError):
        return False
    return len(value) == 10  # fromisoformat nhận cả "20200301"; tập kiểm thử chỉ dùng dạng YYYY-MM-DD


def _case_errors(case, chunk_ids, article_keys):
    errors = [f"{field}={case[field]!r} không hợp lệ"
              for field, allowed in (("type", TYPES), ("domain", DOMAINS), ("source", SOURCES))
              if case[field] not in allowed]
    if not str(case["question"]).strip():
        errors.append("question rỗng")
    if case["event_date"] is not None and not is_iso_date(case["event_date"]):
        errors.append(f"event_date={case['event_date']!r} không phải YYYY-MM-DD")
    if case["type"] == "temporal" and case["event_date"] is None:
        errors.append("câu temporal phải có event_date")
    gold = case["gold_citations"]
    if not isinstance(gold, list):
        return errors + ["gold_citations phải là danh sách"]
    if case["type"] == "out_of_scope" and gold:
        errors.append("câu out_of_scope phải có gold_citations rỗng")
    if case["type"] != "out_of_scope" and not gold:
        errors.append("thiếu gold_citations")
    return errors + [f"căn cứ {g} không có trong kho" for g in gold if g not in chunk_ids and g not in article_keys]


def validate_cases(cases, chunk_ids):
    """Danh sách lỗi '<id>: <mô tả>'; rỗng nghĩa là hợp lệ. chunk_ids: mọi chunk_id trong chunks.jsonl."""
    chunk_ids = set(chunk_ids)
    article_keys = {article_key(c) for c in chunk_ids}
    errors, seen = [], set()
    for i, case in enumerate(cases):
        cid = case.get("id", f"#{i}")
        missing = [f for f in FIELDS if f not in case]
        if missing:
            errors.append(f"{cid}: thiếu trường {', '.join(missing)}")
            continue
        if cid in seen:
            errors.append(f"{cid}: id trùng")
        seen.add(cid)
        errors += [f"{cid}: {e}" for e in _case_errors(case, chunk_ids, article_keys)]
    return errors
```

```python
def _law_ids(laws):
    return {info["number"].upper(): law_id for law_id, info in laws.items()}


def _case(case_id, question, law_id, gold, laws, source, reference_answer=""):
    return {"id": case_id, "question": question.strip(), "type": "situational", "domain": laws[law_id]["domain"],
            "event_date": None, "gold_citations": gold, "reference_answer": reference_answer, "source": source,
            "checked_by": None}


def zalo_cases(queries, qrels, laws, exclude=frozenset()):
    """Câu Zalo có mọi căn cứ thuộc kho, căn cứ ở cấp Điều. qrels: [(query_id, '91/2015/qh13+321')]."""
    ids = _law_ids(laws)
    refs = collections.defaultdict(list)
    for qid, corpus_id in qrels:
        number, article = corpus_id.rsplit("+", 1)
        refs[qid].append((ids.get(number.upper()), article))
    out = []
    for qid in sorted(refs):
        case_id = f"Z_{qid[:8]}"
        if qid not in queries or case_id in exclude or any(law_id is None for law_id, _ in refs[qid]):
            continue
        gold = sorted({f"{law_id}_Dieu{article}" for law_id, article in refs[qid]})
        out.append(_case(case_id, queries[qid], refs[qid][0][0], gold, laws, "zalo"))
    return out


def _synthetic_gold(law_id, ground_truth):
    clause_gold = [f"{law_id}_Dieu{a}_Khoan{k}" for a, _, k in (c.partition(".") for c in ground_truth["clauses"]) if k]
    covered = {article_key(g) for g in clause_gold}
    return clause_gold + [f"{law_id}_Dieu{a}" for a in ground_truth["articles"] if f"{law_id}_Dieu{a}" not in covered]


def synthetic_cases(raw_cases, laws):
    """Câu tổng hợp trong data/test_cases.json: ground_truth {articles: ["3"], clauses: ["3.1"]} → chunk_id."""
    ids = _law_ids(laws)
    out = []
    for t in raw_cases:
        law_id = ids.get(t["law_number"].upper())
        if law_id is None:
            raise ValueError(f"{t['id']}: số hiệu {t['law_number']} chưa có trong laws.json")
        out.append(_case(t["id"], t["question"], law_id, _synthetic_gold(law_id, t["ground_truth"]), laws,
                         "synthetic", t.get("expected_answer", "")))
    return out


def read_exclude(path):
    """Mã câu Zalo bị loại khi duyệt; mỗi dòng '<id>  # lý do'."""
    path = Path(path)
    if not path.exists():
        return frozenset()
    lines = (line.split("#", 1)[0].strip() for line in path.read_text(encoding="utf-8").splitlines())
    return frozenset(line for line in lines if line)


def load_manual(path):
    path = Path(path)
    if not path.exists():
        return []
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except json.JSONDecodeError as e:
        sys.exit(f"{path} không phải JSON hợp lệ: {e}")


def load_zalo(zalo_dir):
    zalo_dir = Path(zalo_dir)
    queries = {r["_id"]: r["text"] for r in read_jsonl(zalo_dir / "queries.jsonl")}
    qrels = [(r["query-id"], r["corpus-id"]) for split in ("train", "test")
             for r in read_jsonl(zalo_dir / "qrels" / f"{split}.jsonl")]
    return queries, qrels


def main():
    argparse.ArgumentParser(description=__doc__).parse_args()
    laws = json.loads(LAWS_PATH.read_text(encoding="utf-8"))
    synthetic_raw = json.loads(SYNTHETIC_PATH.read_text(encoding="utf-8"))["test_cases"]
    cases = (zalo_cases(*load_zalo(ZALO_DIR), laws, read_exclude(EXCLUDE_PATH))
             + synthetic_cases(synthetic_raw, laws) + load_manual(MANUAL_PATH))
    errors = validate_cases(cases, [c["chunk_id"] for c in read_jsonl(CHUNKS_PATH)])
    if errors:
        print("\n".join(errors), file=sys.stderr)
        sys.exit(f"{len(errors)} lỗi, chưa ghi {OUT_PATH}")
    OUT_PATH.write_text(json.dumps(cases, ensure_ascii=False, indent=1) + "\n", encoding="utf-8")
    print(f"đã ghi {len(cases)} câu vào {OUT_PATH}")
    for field in ("source", "type", "domain"):
        print(f"  {field}: {dict(collections.Counter(c[field] for c in cases))}")


if __name__ == "__main__":
    main()
```

```python
def build_index(chunks, embed, db_path=DB_PATH, rebuild=False, collection=COLLECTION, with_header=True):
    client = chromadb.PersistentClient(path=str(db_path))
    if rebuild and collection in [c.name for c in client.list_collections()]:
        client.delete_collection(collection)
    col = client.get_or_create_collection(collection, metadata={"hnsw:space": "cosine"})
    for i in range(0, len(chunks), BATCH):
        batch = chunks[i:i + BATCH]
        col.upsert(
            ids=[c["chunk_id"] for c in batch],
            embeddings=embed([embedding_text(c) if with_header else c["content"] for c in batch]),
            documents=[c["content"] for c in batch],
            metadatas=[chunk_metadata(c) for c in batch],
        )
    return col


def main():
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--variant", choices=VARIANTS, default="clause_header")
    ap.add_argument("--rebuild", action="store_true", help="xóa collection cũ trước khi nạp")
    ap.add_argument("--query", help="truy vấn thử sau khi nạp")
    args = ap.parse_args()

    file, collection, with_header = VARIANTS[args.variant]
    chunks = read_jsonl(PROCESSED_DIR / file)
    embed = make_embedder()
    start = time.perf_counter()
    col = build_index(chunks, embed, rebuild=args.rebuild, collection=collection, with_header=with_header)
    print(f"đã nạp {col.count()} chunk vào {DB_PATH}/{collection} trong {time.perf_counter() - start:.0f}s")
```

```python
"""Lọc đoạn văn bản theo khoảng hiệu lực tại một ngày, dùng chung cho BM25 (Python) và Chroma (where)."""
from datetime import date as _date

from src.ingestion.index import OPEN_END


def check_date(value):
    """None hoặc ngày ISO đã chuẩn hóa; sai định dạng thì ValueError (ngày có thể do LLM trích ra)."""
    return None if value is None else _date.fromisoformat(value).isoformat()


def is_valid_at(chunk, date=None):
    """date=None: chỉ văn bản còn hiệu lực; có date: effective_date <= date < expiry_date."""
    if date is None:
        return chunk.get("expiry_date") is None
    date = check_date(date)
    expiry = chunk.get("expiry_date")
    return chunk["effective_date"] <= date and (expiry is None or date < expiry)


def chroma_where(date=None):
    if date is None:
        return {"effective_to": {"$eq": OPEN_END}}
    d = int(check_date(date).replace("-", ""))
    return {"$and": [{"effective_from": {"$lte": d}}, {"effective_to": {"$gt": d}}]}
```

```python
"""Tìm kiếm từ khóa BM25 trên từ đã tách bằng underthesea (từ ghép nối bằng _)."""
import re

from rank_bm25 import BM25Okapi
from underthesea import word_tokenize

from src.retrieval.filters import check_date, is_valid_at

_PUNCT_RE = re.compile(r"[^\w\s]")


def tokenize(text):
    return _PUNCT_RE.sub(" ", word_tokenize(text, format="text").lower()).split()


class BM25Index:
    # shortcut: dựng lại chỉ mục mỗi lần khởi động (~10 giây cho 4 luật); lưu ra đĩa nếu kho lớn hơn nhiều
    def __init__(self, chunks, text_fn=lambda c: c["content"]):
        self.chunks = chunks
        self.bm25 = BM25Okapi([tokenize(text_fn(c)) for c in chunks])

    def search(self, query, k=50, date=None):
        date = check_date(date)
        tokens = tokenize(query)
        if not tokens:
            return []
        scores = self.bm25.get_scores(tokens)
        valid = (i for i, c in enumerate(self.chunks) if is_valid_at(c, date) and scores[i] > 0)
        ranked = sorted(valid, key=lambda i: -scores[i])[:k]
        return [(self.chunks[i]["chunk_id"], float(scores[i])) for i in ranked]
```

```python
"""Truy xuất căn cứ: BM25, Dense, Hybrid (RRF) và Hybrid + Cross-Encoder rerank, lọc theo khoảng hiệu lực."""
import chromadb

from src.ingestion.index import DB_PATH, PROCESSED_DIR, VARIANTS, embedding_text, make_embedder
from src.ingestion.legal_chunker import article_chunks, article_key, read_jsonl
from src.retrieval.bm25 import BM25Index
from src.retrieval.dense import DenseIndex
from src.retrieval.filters import check_date

MODES = ("bm25", "dense", "hybrid", "hybrid_rerank")
RRF_K = 60
CANDIDATES = 50  # số ứng viên lấy từ mỗi nhánh BM25 / Dense trước khi hợp nhất
RERANK_TOP = 20  # số ứng viên đưa vào reranker; giảm còn 10 nếu CPU > 3 giây/câu
RERANK_MODEL = "BAAI/bge-reranker-v2-m3"


def rrf(rankings, k=RRF_K):
    """Reciprocal Rank Fusion: điểm = tổng 1/(k + hạng) qua các danh sách."""
    scores = {}
    for ranking in rankings:
        for rank, cid in enumerate(ranking, 1):
            scores[cid] = scores.get(cid, 0.0) + 1.0 / (k + rank)
    return sorted(scores.items(), key=lambda item: -item[1])


def make_reranker(model=RERANK_MODEL):
    from sentence_transformers import CrossEncoder
    encoder = CrossEncoder(model, max_length=512)
    return lambda query, texts: [float(s) for s in encoder.predict([(query, t) for t in texts])]


class Retriever:
    def __init__(self, chunks, bm25, dense, rerank=None):
        self.by_id = {c["chunk_id"]: c for c in chunks}
        self.bm25, self.dense, self.rerank = bm25, dense, rerank

    def search(self, query, k=5, date=None, mode="hybrid_rerank"):
        """Top-k chunk (kèm 'score'); date=None chỉ lấy văn bản còn hiệu lực."""
        if mode not in MODES:
            raise ValueError(f"mode phải thuộc {MODES}, nhận {mode!r}")
        date = check_date(date)
        if not query.strip():
            return []
        if mode == "bm25":
            ranked = self.bm25.search(query, k, date)
        elif mode == "dense":
            ranked = self.dense.search(query, k, date)
        else:
            fused = rrf([[cid for cid, _ in index.search(query, CANDIDATES, date)] for index in (self.bm25, self.dense)])
            ranked = fused[:k] if mode == "hybrid" else self._rerank(query, fused[:RERANK_TOP])[:k]
        stale = [cid for cid, _ in ranked if cid not in self.by_id]
        if stale:
            raise RuntimeError(f"chỉ mục lệch với file chunk ({stale[0]}): chạy lại "
                               "python -m src.ingestion.index --variant <biến thể> --rebuild")
        return [{**self.by_id[cid], "score": score} for cid, score in ranked]

    def _rerank(self, query, candidates):
        if self.rerank is None:
            raise RuntimeError("Retriever chưa có reranker cho mode hybrid_rerank")
        ids = [cid for cid, _ in candidates if cid in self.by_id]
        scores = self.rerank(query, [embedding_text(self.by_id[cid]) for cid in ids]) if ids else []
        return sorted(zip(ids, scores), key=lambda item: -item[1])

    def get_article(self, law_id, number):
        """Toàn văn một Điều (dẫn nhập + các Khoản theo thứ tự); None nếu không có."""
        key = f"{law_id}_Dieu{number}"
        parts = [c for cid, c in self.by_id.items() if article_key(cid) == key]
        return article_chunks(parts)[0] if parts else None


def load_retriever(variant="clause_header", embed=None, rerank=None, db_path=DB_PATH):
    """Retriever trên một biến thể đã nạp bằng `python -m src.ingestion.index --variant <variant>`."""
    file, collection, with_header = VARIANTS[variant]
    chunks = read_jsonl(PROCESSED_DIR / file)
    col = chromadb.PersistentClient(path=str(db_path)).get_collection(collection)
    text_fn = embedding_text if with_header else (lambda c: c["content"])
    return Retriever(chunks, BM25Index(chunks, text_fn), DenseIndex(col, embed or make_embedder()),
                     rerank or make_reranker())
```

Lưu ý: `_rerank` bỏ qua id lạ để reranker không lỗi; `search` vẫn bắt id lạ ở các mode khác bằng kiểm tra `stale`. Với hybrid_rerank, id lạ bị bỏ qua im lặng — chấp nhận vì cùng lỗi sẽ lộ ở mode bm25/dense trong RQ1.

```python
"""Chỉ số truy xuất: Hit@k, Recall@k, MRR@k (cấp Điều hoặc Khoản) và DRM@k."""
from src.ingestion.legal_chunker import article_key


def article_ranking(chunks):
    """Khóa Điều theo thứ tự xuất hiện, bỏ trùng; đoạn cố định góp mọi Điều nó phủ qua."""
    keys = []
    for c in chunks:
        found = ([f"{c['law_id']}_Dieu{a.removeprefix('Điều ')}" for a in c["articles"]] if "articles" in c
                 else [article_key(c["chunk_id"])])
        keys = keys + [k for k in found if k not in keys]
    return keys


def hit_at(ranked, gold, k):
    return float(bool(set(ranked[:k]) & set(gold)))


def recall_at(ranked, gold, k):
    return len(set(ranked[:k]) & set(gold)) / len(set(gold))


def mrr_at(ranked, gold, k):
    gold = set(gold)
    return next((1.0 / rank for rank, key in enumerate(ranked[:k], 1) if key in gold), 0.0)


def drm_at(chunks, gold, k):
    """Tỷ lệ đoạn trong top-k thuộc văn bản không có trong căn cứ chuẩn; None nếu không có kết quả."""
    top = chunks[:k]
    if not top:
        return None
    gold_laws = {g.split("_", 1)[0] for g in gold}
    return sum(c["law_id"] not in gold_laws for c in top) / len(top)


def mean(values):
    values = [v for v in values if v is not None]
    return sum(values) / len(values) if values else None
```

```python
"""Thực nghiệm truy xuất: RQ1 (phương pháp) và RQ2 (phân đoạn) trên các câu situational, tách theo nguồn."""
import argparse
import csv
import functools
import json
import sys
import time
from pathlib import Path

from eval.metrics import article_ranking, drm_at, hit_at, mean, mrr_at, recall_at
from src.ingestion.index import make_embedder
from src.ingestion.legal_chunker import article_key
from src.retrieval.retriever import MODES, load_retriever, make_reranker

CASES_PATH = Path("eval/test_cases.json")
RESULTS_DIR = Path("eval/results")
DEPTH = 30  # số chunk lấy về để dựng xếp hạng cấp Điều (một Điều có thể chiếm nhiều chunk)
CONFIGS = {
    "rq1": [("clause_header", mode) for mode in MODES],
    "rq2": [(variant, "hybrid_rerank") for variant in ("fixed", "article", "clause_plain", "clause_header")],
}
CLAUSE_METRICS = ("hit5_clause", "recall10_clause", "mrr10_clause")
METRICS = ("hit5", "recall10", "mrr10", "drm5", *CLAUSE_METRICS, "ms")


def eval_case(retriever, case, mode, clause_level):
    start = time.perf_counter()
    chunks = retriever.search(case["question"], k=DEPTH, date=case["event_date"], mode=mode)
    ms = round((time.perf_counter() - start) * 1000)
    gold = case["gold_citations"]
    gold_articles = sorted({article_key(g) for g in gold})
    ranked = article_ranking(chunks)
    row = {"id": case["id"], "source": case["source"], "domain": case["domain"],
           "hit5": hit_at(ranked, gold_articles, 5), "recall10": recall_at(ranked, gold_articles, 10),
           "mrr10": mrr_at(ranked, gold_articles, 10), "drm5": drm_at(chunks, gold_articles, 5), "ms": ms}
    if not (clause_level and all("_Khoan" in g for g in gold)):
        return {**row, **dict.fromkeys(CLAUSE_METRICS)}
    ids = [c["chunk_id"] for c in chunks]
    return {**row, "hit5_clause": hit_at(ids, gold, 5), "recall10_clause": recall_at(ids, gold, 10),
            "mrr10_clause": mrr_at(ids, gold, 10)}


def run(rq, cases, get_retriever):
    return [{"config": f"{variant}/{mode}", **eval_case(get_retriever(variant), c, mode, variant.startswith("clause"))}
            for variant, mode in CONFIGS[rq] for c in cases]


def summarize(rows):
    """Một dòng cho mỗi (config, source) và (config, 'all'); trung bình bỏ qua giá trị None."""
    groups = {}
    for r in rows:
        for source in (r["source"], "all"):
            groups.setdefault((r["config"], source), []).append(r)
    return [{"config": cfg, "source": src, "n": len(rs), **{m: mean(r[m] for r in rs) for m in METRICS}}
            for (cfg, src), rs in groups.items()]


def _fmt(value):
    return "–" if value is None else f"{value:.3f}"


def to_markdown(summary):
    head = ["| config | source | n | Hit@5 | Recall@10 | MRR@10 | DRM@5 | Hit@5 (Khoản) | ms |",
            "|---|---|---|---|---|---|---|---|---|"]
    lines = [f"| {s['config']} | {s['source']} | {s['n']} | {_fmt(s['hit5'])} | {_fmt(s['recall10'])} | "
             f"{_fmt(s['mrr10'])} | {_fmt(s['drm5'])} | {_fmt(s['hit5_clause'])} | {s['ms']:.0f} |" for s in summary]
    return "\n".join(head + lines) + "\n"


def main():
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--rq", choices=sorted(CONFIGS), required=True)
    ap.add_argument("--limit", type=int, help="chỉ chạy N câu đầu để thử")
    args = ap.parse_args()
    cases = [c for c in json.loads(CASES_PATH.read_text(encoding="utf-8")) if c["type"] == "situational"]
    cases = cases[: args.limit]
    if not cases:
        sys.exit(f"không có câu situational trong {CASES_PATH}: chạy python -m eval.import_cases trước")
    embed, rerank = make_embedder(), make_reranker()
    get_retriever = functools.cache(lambda variant: load_retriever(variant, embed, rerank))
    rows = run(args.rq, cases, get_retriever)
    RESULTS_DIR.mkdir(parents=True, exist_ok=True)
    with (RESULTS_DIR / f"retrieval_{args.rq}.csv").open("w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)
    md = to_markdown(summarize(rows))
    (RESULTS_DIR / f"retrieval_{args.rq}.md").write_text(md, encoding="utf-8")
    print(md)


if __name__ == "__main__":
    main()
```
