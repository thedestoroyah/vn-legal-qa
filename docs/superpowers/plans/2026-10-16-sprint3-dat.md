**Goal:** Tác tử LangGraph chạy được qua dòng lệnh với đủ các nhánh (out_of_scope, direct, situational có vòng viết lại, verify gỡ trích dẫn sai), có 5 cấu hình RQ3. Kết quả trả về đúng định dạng Phụ lục A.5. Đã chạy thử 10 câu thật với Gemini.

**Architecture:**
- `src/generation/` có thêm `prompts.py` (4 prompt tiếng Việt) và `citations.py` (hàm thuần tách, gỡ và gắn nhãn `[chunk_id]`).
- `src/agent/` gồm 4 file:
  - `state.py`: trạng thái và cấu hình.
  - `nodes.py`: mỗi nút là một hàm `(state, deps) -> dict`.
  - `graph.py`: dựng StateGraph và ghi trace.
  - `run.py`: lớp `Agent` và CLI.
- Tác tử chỉ gọi `Retriever.search`/`get_article` và `LLM.chat`/`chat_json`. Test dùng LLM thật với client giả, trả lời theo từng nút; Retriever cũng là bản giả.

```python
"""Client LLM chuẩn OpenAI (Gemini/Groq đổi bằng .env), cache xuống đĩa, đếm lượt gọi, giãn cách theo RPM."""
import hashlib
import json
import os
import time
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
ENV_PATH = ROOT / ".env"  # gốc repo: chạy lệnh từ thư mục nào (vd src/ui) cũng nạp đúng .env
CACHE_DIR = ROOT / ".cache" / "llm"
DEFAULT_MAX_RETRIES = 8  # SDK openai tự chờ lũy tiến và thử lại khi gặp 429/5xx
JSON_RETRY_HINT = "Chỉ trả về một đối tượng JSON hợp lệ, không kèm giải thích."
_DECODER = json.JSONDecoder()


class LLM:
    def __init__(self, client, model, cache_dir=CACHE_DIR, base_url="", min_interval=0.0):
        self.client = client
        self.model = model
        self.cache_dir = Path(cache_dir)
        self.base_url = base_url
        self.min_interval = min_interval  # giây tối thiểu giữa 2 lượt gọi thật; Gemini free tier 15 RPM → 4.2
        self.calls = 0  # lượt gọi API thật, không tính cache
        self.requests = 0  # lượt gọi logic, kể cả trúng cache: "lượt gọi LLM/câu" của RQ3 không phụ thuộc cache
        self._last_call = None

    def _cache_path(self, messages, params):
        raw = json.dumps({"base_url": self.base_url, "model": self.model, "messages": messages, "params": params},
                         ensure_ascii=False, sort_keys=True)
        return self.cache_dir / f"{hashlib.sha256(raw.encode()).hexdigest()}.json"

    def _throttle(self):
        if self._last_call is not None:
            wait = self.min_interval - (time.monotonic() - self._last_call)
            if wait > 0:
                time.sleep(wait)
        self._last_call = time.monotonic()

    def chat(self, messages, temperature=0.0, **params):
        self.requests += 1
        params = {"temperature": temperature, **params}
        path = self._cache_path(messages, params)
        try:
            return json.loads(path.read_text(encoding="utf-8"))["content"]
        except (FileNotFoundError, json.JSONDecodeError, KeyError):
            pass  # chưa có hoặc file cache hỏng: gọi API
        self._throttle()
        resp = self.client.chat.completions.create(model=self.model, messages=messages, **params)
        self.calls += 1
        content = (resp.choices[0].message.content if resp.choices else None) or ""
        if content:  # không cache phản hồi rỗng (bị chặn, lỗi tạm) để lần sau còn gọi lại
            path.parent.mkdir(parents=True, exist_ok=True)
            tmp = path.with_suffix(f".{os.getpid()}.tmp")  # mỗi tiến trình một file tạm: chạy song song không ghi đè nhau
            tmp.write_text(json.dumps({"content": content}, ensure_ascii=False), encoding="utf-8")
            tmp.replace(path)  # ghi nguyên tử: dừng giữa chừng không để lại file cache hỏng
        return content

    def chat_json(self, messages, temperature=0.0, **params):
        """Thử lại 1 lần với lời nhắc chỉ trả JSON; vẫn hỏng thì ValueError để nút gọi tự dùng giá trị mặc định."""
        retry = messages + [{"role": "user", "content": JSON_RETRY_HINT}]
        for attempt in (messages, retry):
            parsed = parse_json(self.chat(attempt, temperature, **params))
            if parsed is not None:
                return parsed
            # bỏ bản cache hỏng: lần chạy sau gọi lại API thay vì lặp lại đúng câu trả lời hỏng
            self._cache_path(attempt, {"temperature": temperature, **params}).unlink(missing_ok=True)
        raise ValueError("LLM không trả về JSON hợp lệ sau 2 lần")


def parse_json(text):
    """Đối tượng JSON hợp lệ đầu tiên trong text (bỏ ```json, lời dẫn, ngoặc nhọn lạc); None nếu không có."""
    text = text or ""
    for start, ch in enumerate(text):
        if ch != "{":
            continue
        try:
            value, _ = _DECODER.raw_decode(text, start)
        except json.JSONDecodeError:
            continue
        if isinstance(value, dict):
            return value
    return None


def _load_env_file(path=ENV_PATH):
    """Nạp KEY=VALUE từ .env, không ghi đè biến đã có trong shell."""
    # shortcut: không hỗ trợ export, giá trị nhiều dòng, comment cuối dòng; đổi sang python-dotenv nếu .env phức tạp hơn
    p = Path(path)
    if not p.exists():
        return
    for line in p.read_text(encoding="utf-8").splitlines():
        line = line.strip()
        if line and not line.startswith("#") and "=" in line:
            key, value = line.split("=", 1)
            os.environ.setdefault(key.strip(), value.strip().strip("'\""))


def from_env(prefix="LLM", cache_dir=CACHE_DIR, env_path=ENV_PATH):
    """LLM từ <prefix>_BASE_URL, <prefix>_API_KEY, <prefix>_MODEL (+ _MIN_INTERVAL, _MAX_RETRIES); prefix JUDGE cho bộ chấm."""
    from openai import OpenAI
    _load_env_file(env_path)
    missing = [f"{prefix}_{k}" for k in ("BASE_URL", "API_KEY", "MODEL") if not os.environ.get(f"{prefix}_{k}")]
    if missing:
        raise RuntimeError(f"thiếu biến môi trường: {', '.join(missing)} (xem .env.example)")
    base_url = os.environ[f"{prefix}_BASE_URL"]
    client = OpenAI(base_url=base_url, api_key=os.environ[f"{prefix}_API_KEY"],
                    max_retries=int(os.environ.get(f"{prefix}_MAX_RETRIES", DEFAULT_MAX_RETRIES)))
    return LLM(client, os.environ[f"{prefix}_MODEL"], cache_dir, base_url,
               float(os.environ.get(f"{prefix}_MIN_INTERVAL", 0)))
```

```python
"""Tách, gỡ và gắn nhãn trích dẫn [chunk_id] trong câu trả lời (bước verify, spec mục 5)."""
import re

CITE_RE = re.compile(r"[ \t]*\[([^\[\]]+)\]")
ID_RE = re.compile(r"[A-Z]+\d{4}_Dieu\w+")
NO_BASIS = "(chưa có căn cứ)"
_SEP_RE = re.compile(r"[,;\s]+")
_SPLIT_RE = re.compile(r"((?<=[.!?])\s+|\n+)")
_END_RE = re.compile(r"([.!?]*)$")


def _ids(inner):
    return [t for t in _SEP_RE.split(inner.strip()) if ID_RE.fullmatch(t)]


def cited_ids(text):
    """Mã trích dẫn theo thứ tự xuất hiện, không trùng; ngoặc không chứa mã (vd [1]) bị bỏ qua."""
    return list(dict.fromkeys(cid for m in CITE_RE.finditer(text) for cid in _ids(m.group(1))))


def drop_citations(text, bad):
    """Gỡ các mã trong `bad`; ngoặc không còn mã nào thì gỡ cả ngoặc."""
    def fix(m):
        ids = _ids(m.group(1))
        keep = [cid for cid in ids if cid not in bad]
        if keep == ids:
            return m.group(0)  # không có gì để gỡ (kể cả ngoặc thường như [1])
        return f" [{', '.join(keep)}]" if keep else ""
    return CITE_RE.sub(fix, text)


def strip_unknown(text, known):
    """verify(a): gỡ mã không có trong ngữ cảnh. Trả (văn bản, các mã đã gỡ)."""
    removed = [cid for cid in cited_ids(text) if cid not in known]
    return drop_citations(text, set(removed)), removed


def sentences(text):
    """Phần tử chẵn là câu, phần tử lẻ là khoảng trắng phân cách; ''.join(sentences(t)) == t."""
    return _SPLIT_RE.split(text)


def cited_sentences(text):
    """[(chỉ số trong sentences(text), câu, các mã)] cho các câu có trích dẫn."""
    parts = sentences(text)
    return [(i, parts[i], cited_ids(parts[i])) for i in range(0, len(parts), 2) if cited_ids(parts[i])]


def drop_pairs(text, verdicts):
    """Gỡ các cặp {chỉ số câu: {mã}} mà bước kiểm chứng LLM cho là không được hỗ trợ."""
    return "".join(drop_citations(p, verdicts[i]) if i in verdicts else p for i, p in enumerate(sentences(text)))


def lost_sentences(before, after):
    """Chỉ số các câu có trích dẫn trong `before` nhưng không còn mã nào trong `after`."""
    old, new = sentences(before), sentences(after)
    return [i for i in range(0, min(len(old), len(new)), 2) if cited_ids(old[i]) and not cited_ids(new[i])]


def label_sentences(text, indices):
    """Gắn NO_BASIS vào cuối các câu được chỉ định, trước dấu kết câu."""
    marked = set(indices)
    return "".join(_END_RE.sub(lambda m: f" {NO_BASIS}{m.group(1)}", p.rstrip(), count=1) if i in marked else p
                   for i, p in enumerate(sentences(text)))
```

def _messages(node, rules, user):
    return [{"role": "system", "content": f"{ROLES[node]}\n\n{rules}"}, {"role": "user", "content": user}]


def law_list(laws):
    return "\n".join(f"- {law_id}: {info['document']} (hiệu lực từ {info['effective_date']}, "
                     + (f"hết hiệu lực từ {info['expiry_date']})" if info.get("expiry_date") else "còn hiệu lực)")
                     for law_id, info in laws.items())


def format_context(chunks):
    """Mỗi đoạn: '[chunk_id] văn bản, Điều (tên), Khoản:' rồi nội dung ở dòng sau."""
    def head(c):
        where = f"{c['article']} ({c['article_title']})" if c.get("article_title") else c["article"]
        return f"[{c['chunk_id']}] {c['document']}, {where}" + (f", {c['clause']}" if c.get("clause") else "")
    return "\n\n".join(f"{head(c)}:\n{c['content']}" for c in chunks)


def route_messages(question, laws):
    # replace thay vì format: luật JSON trong ví dụ có ngoặc nhọn
    return _messages("route", ROUTE_RULES.replace("{laws}", law_list(laws)), f"Câu hỏi: {question}")


def grade_messages(question, evidence, queries):
    return _messages("grade", GRADE_RULES, f"Câu hỏi: {question}\nCác truy vấn đã dùng: {'; '.join(queries)}\n\n"
                                           f"Đoạn trích:\n{format_context(evidence)}")


def generate_messages(question, evidence, event_date=None, unsupported=()):
    parts = [f"Đoạn trích:\n{format_context(evidence)}"]
    if event_date:
        parts.append(f"Sự việc xảy ra ngày {event_date}.")
    parts.append(f"Câu hỏi: {question}")
    if unsupported:
        parts.append("Ở lần trả lời trước, các câu sau không được đoạn trích hỗ trợ; hãy sửa cho đúng đoạn trích "
                     "hoặc bỏ đi:\n" + "\n".join(f"- {s}" for s in unsupported))
    return _messages("generate", GENERATE_RULES, "\n\n".join(parts))


def verify_messages(pairs, by_id):
    items = [f"Câu {n}: {sentence}\n" + "\n".join(f"  [{cid}] {by_id[cid]['content']}" for cid in ids)
             for n, (_, sentence, ids) in enumerate(pairs, 1)]
    return _messages("verify", VERIFY_RULES, "\n\n".join(items))

"""LLM và Retriever giả cho test tác tử: LLM thật (cache, chat_json) trên client trả lời theo từng nút."""
import json
import re
from types import SimpleNamespace

from src.agent.nodes import Deps
from src.generation.llm import LLM
from src.generation.prompts import ROLES

LAWS = {
    "BLDS2015": {"document": "Bộ luật Dân sự 2015", "effective_date": "2017-01-01", "expiry_date": None},
    "BLLD2012": {"document": "Bộ luật Lao động 2012", "effective_date": "2013-05-01", "expiry_date": "2021-01-01"},
    "BLLD2019": {"document": "Bộ luật Lao động 2019", "effective_date": "2021-01-01", "expiry_date": None},
}


def node_of(messages):
    system = messages[0]["content"]
    return next(name for name, role in ROLES.items() if system.startswith(role))


class ScriptedClient:
    """Client openai giả: mỗi nút có danh sách trả lời riêng, lấy lần lượt; nút gọi ngoài kịch bản thì KeyError."""

    def __init__(self, **replies):
        self.replies = {node: list(items) for node, items in replies.items()}
        self.seen = []
        self.chat = SimpleNamespace(completions=SimpleNamespace(create=self._create))

    def _create(self, model, messages, **params):
        node = node_of(messages)
        self.seen.append((node, messages))
        reply = self.replies[node].pop(0)
        content = reply if isinstance(reply, str) else json.dumps(reply, ensure_ascii=False)
        return SimpleNamespace(choices=[SimpleNamespace(message=SimpleNamespace(content=content))])


def make_llm(tmp_path, **replies):
    return LLM(ScriptedClient(**replies), "fake", tmp_path)


def nodes_called(llm):
    return [node for node, _ in llm.client.seen]


def chunk(cid, content="nội dung"):
    m = re.fullmatch(r"([A-Z]+\d{4})_Dieu(\w+?)(?:_Khoan(\d+))?", cid)
    return {"chunk_id": cid, "law_id": m[1], "document": f"Văn bản {m[1]}", "article": f"Điều {m[2]}",
            "article_title": "Tiêu đề", "clause": f"Khoản {m[3]}" if m[3] else None, "content": content, "score": 0.5}


class FakeRetriever:
    """Mỗi lượt search lấy danh sách kế tiếp trong `results` (hết thì trả rỗng); get_article tra trong `articles`."""

    def __init__(self, results=(), articles=None):
        self.results = list(results)
        self.articles = articles or {}
        self.searches, self.lookups = [], []

    def search(self, query, k=5, date=None, mode="hybrid_rerank"):
        self.searches.append({"query": query, "k": k, "date": date})
        return self.results.pop(0) if self.results else []

    def get_article(self, law_id, number):
        self.lookups.append((law_id, number))
        return self.articles.get(f"{law_id}_Dieu{number}")


def make_deps(tmp_path, retriever=None, **replies):
    return Deps(make_llm(tmp_path, **replies), retriever or FakeRetriever(), LAWS)

"""Các nút của tác tử (spec mục 5). Mỗi nút nhận state và Deps, trả về phần state cần cập nhật."""
import re
from dataclasses import dataclass

from src.agent.state import MAX_EVIDENCE, TOP_K
from src.generation import prompts
from src.retrieval.filters import check_date

ROUTE_TYPES = {"situational", "direct", "out_of_scope"}
REFUSALS = {
    "out_of_scope": "Câu hỏi nằm ngoài phạm vi hệ thống. Hệ thống chỉ hỗ trợ Bộ luật Dân sự, Bộ luật Lao động, "
                    "Luật Doanh nghiệp và Luật Thương mại.",
    "no_evidence": "Không tìm thấy văn bản có hiệu lực tại thời điểm của tình huống trong kho dữ liệu.",
    "insufficient": "Không đủ căn cứ trong kho dữ liệu để trả lời câu hỏi này.",
}


@dataclass(frozen=True)
class Deps:
    llm: object  # src.generation.llm.LLM
    retriever: object  # có search(query, k, date) và get_article(law_id, number)
    laws: dict  # nội dung data/raw/laws.json


def _safe_date(value):
    try:
        return check_date(value)
    except (TypeError, ValueError):
        return None  # ngày LLM trả sai dạng: coi như không có, truy xuất theo văn bản còn hiệu lực


def _law_id(raw, laws):
    """Mã LLM trả ('BLLĐ 2012', 'blds-2015') về khóa trong laws.json; None nếu không khớp."""
    key = re.sub(r"[\s\-_.]", "", str(raw or "")).upper().replace("Đ", "D")
    return key if key in laws else None


def _direct_ref(out, laws):
    law_id = _law_id(out.get("law_id"), laws)
    article = re.search(r"\d+[a-z]?", str(out.get("article") or ""))
    return {"law_id": law_id, "article": article.group(0)} if law_id and article else None


def route(state, deps):
    try:
        out = deps.llm.chat_json(prompts.route_messages(state["question"], deps.laws))
    except ValueError:
        out = {}  # JSON hỏng sau 2 lần: mặc định situational (spec mục 6)
    kind = out.get("type") if out.get("type") in ROUTE_TYPES else "situational"
    return {"route_type": kind, "event_date": _safe_date(out.get("event_date")),
            "direct_ref": _direct_ref(out, deps.laws) if kind == "direct" else None}


def lookup_article(state, deps):
    ref = state["direct_ref"]
    article = deps.retriever.get_article(ref["law_id"], ref["article"])
    return {"evidence": [article] if article else []}


def _merge(old, new):
    """Gộp bằng chứng các vòng (giữ thứ tự, bỏ trùng): vòng viết lại không làm mất đoạn đã tìm được."""
    seen = {c["chunk_id"] for c in old}
    return (old + [c for c in new if c["chunk_id"] not in seen])[:MAX_EVIDENCE]


def retrieve(state, deps):
    # không lọc theo lĩnh vực: router đoán sai lĩnh vực không kéo theo truy xuất sai (spec mục 5)
    found = deps.retriever.search(state["queries"][-1], k=TOP_K, date=state["event_date"])
    return {"queries": state["queries"], "evidence": _merge(state["evidence"], found)}


def refuse(state, deps):
    if state["route_type"] == "out_of_scope":
        reason = "out_of_scope"
    elif not state["evidence"]:
        reason = "no_evidence"
    else:
        reason = "insufficient"
    return {"answer": REFUSALS[reason], "refused": True, "citations": [], "refuse_reason": reason}

def _is_false(value):
    return value is False or str(value).strip().lower() in {"false", "0", "no"}


def grade(state, deps):
    try:
        out = deps.llm.chat_json(prompts.grade_messages(state["question"], state["evidence"], state["queries"]))
    except ValueError:
        return {"sufficient": True, "rewrite": False}  # JSON hỏng: mặc định đủ (spec mục 6)
    if not _is_false(out.get("sufficient")):
        return {"sufficient": True, "rewrite": False}
    new_query = str(out.get("new_query") or "").strip()
    n = state["n_rewrites"]
    if n >= MAX_REWRITES or not new_query or new_query in state["queries"]:
        return {"sufficient": False, "rewrite": False}  # truy vấn trùng/rỗng: viết lại nữa cũng vô ích
    return {"sufficient": False, "rewrite": True, "queries": state["queries"] + [new_query], "n_rewrites": n + 1}


def generate(state, deps):
    messages = prompts.generate_messages(state["question"], state["evidence"], state["event_date"],
                                         state["unsupported"])
    text = deps.llm.chat(messages).strip().removeprefix("Trả lời:").strip()
    n = state["n_generations"] + 1
    if not text or prompts.INSUFFICIENT in text:
        return {"answer": "", "insufficient": True, "n_generations": n}
    return {"answer": text, "insufficient": False, "n_generations": n}


def _unsupported_pairs(pairs, by_id, deps):
    """{chỉ số câu: {mã}} các cặp LLM cho là không được hỗ trợ; mục sai dạng bị bỏ qua, JSON hỏng thì không gỡ gì."""
    try:
        out = deps.llm.chat_json(prompts.verify_messages(pairs, by_id))
    except ValueError:
        return {}
    items = out.get("unsupported")
    verdicts = {}
    for item in items if isinstance(items, list) else []:
        if not isinstance(item, dict):
            continue
        try:
            n = int(item.get("sentence"))
        except (TypeError, ValueError):
            continue
        if 1 <= n <= len(pairs) and item.get("chunk_id") in pairs[n - 1][2]:
            verdicts.setdefault(pairs[n - 1][0], set()).add(item["chunk_id"])
    return verdicts


def _citation(chunk):
    return {key: chunk.get(key) for key in ("chunk_id", "document", "article", "clause", "content")}


def verify(state, deps, use_llm=True):
    """(a) gỡ mã ngoài ngữ cảnh; (b) LLM kiểm từng cặp (câu, đoạn). Câu mất hết căn cứ: sinh lại 1 lần rồi gắn nhãn."""
    draft = state["answer"]
    by_id = {c["chunk_id"]: c for c in state["evidence"]}
    answer, removed = citations.strip_unknown(draft, by_id)
    if use_llm:
        pairs = citations.cited_sentences(answer)
        verdicts = _unsupported_pairs(pairs, by_id, deps) if pairs else {}
        removed = removed + [cid for ids in verdicts.values() for cid in sorted(ids)]
        answer = citations.drop_pairs(answer, verdicts)
    lost = citations.lost_sentences(draft, answer)
    if lost and use_llm and state["n_generations"] < 2:
        parts = citations.sentences(draft)
        return {"answer": answer, "removed": removed, "regenerate": True, "unsupported": [parts[i] for i in lost]}
    answer = citations.label_sentences(answer, lost)
    ids = citations.cited_ids(answer)
    if not ids:
        answer = f"{answer}\n\n{NO_CITATION_NOTE}"
    return {"answer": answer, "removed": removed, "regenerate": False, "unsupported": [],
            "citations": [_citation(by_id[cid]) for cid in ids], "refused": False}

```python
"""Đồ thị LangGraph của tác tử (spec mục 5) và 5 cấu hình RQ3; mỗi nút ghi một bước trace."""
import time
from functools import partial

from langgraph.graph import END, START, StateGraph

from src.agent import nodes
from src.agent.state import CONFIGS, AgentState

_BULKY = ("evidence", "answer", "citations")


def _summary(update):
    """Đầu ra gọn cho trace: bỏ văn bản dài, bằng chứng chỉ giữ chunk_id."""
    out = {k: v for k, v in update.items() if k not in _BULKY}
    if "evidence" in update:
        out["evidence"] = [c["chunk_id"] for c in update["evidence"]]
    return out


def _traced(name, fn, deps):
    def step(state):
        calls, start = deps.llm.requests, time.perf_counter()
        update = fn(state, deps)
        record = {"node": name, "output": _summary(update), "ms": round((time.perf_counter() - start) * 1000),
                  "llm_calls": deps.llm.requests - calls}
        return {**update, "trace": [record]}
    return step


def _after_route(state):
    if state["route_type"] == "out_of_scope":
        return "refuse"
    return "lookup_article" if state["route_type"] == "direct" and state["direct_ref"] else "retrieve"


def _after_grade(state):
    if state["sufficient"]:
        return "generate"
    return "retrieve" if state["rewrite"] else "refuse"


def build_graph(deps, config="full"):
    flags = CONFIGS[config]
    steps = {"route": nodes.route, "lookup_article": nodes.lookup_article, "retrieve": nodes.retrieve,
             "grade": nodes.grade, "generate": nodes.generate, "refuse": nodes.refuse,
             "verify": partial(nodes.verify, use_llm=flags["use_verify"])}
    graph = StateGraph(AgentState)
    for name, fn in steps.items():
        graph.add_node(name, _traced(name, fn, deps))
    # nút bị cấu hình tắt (route, grade) vẫn có trong đồ thị nhưng không có cạnh nào dẫn tới
    graph.add_conditional_edges(START, lambda s: "route" if flags["use_router"] else "retrieve")
    graph.add_conditional_edges("route", _after_route)
    graph.add_conditional_edges("lookup_article", lambda s: "generate" if s["evidence"] else "retrieve")
    graph.add_conditional_edges(
        "retrieve", lambda s: "refuse" if not s["evidence"] else "grade" if flags["use_grading"] else "generate")
    graph.add_conditional_edges("grade", _after_grade)
    graph.add_conditional_edges("generate", lambda s: "refuse" if s["insufficient"] else "verify")
    graph.add_conditional_edges("verify", lambda s: "generate" if s["regenerate"] else END)
    graph.add_edge("refuse", END)
    return graph.compile()
```

"""Chạy tác tử: Agent.run(question, config) -> dict theo Phụ lục A.5.

CLI: python -m src.agent.run "câu hỏi" [--config full] [--variant clause_header] [--json]
"""
import argparse
import json

from src.agent.graph import build_graph
from src.agent.nodes import Deps
from src.agent.state import CONFIGS, initial_state
from src.ingestion.legal_chunker import LAWS_PATH


class Agent:
    def __init__(self, deps):
        self.deps = deps
        self._graphs = {}  # mỗi cấu hình compile một lần, dùng lại cho mọi câu hỏi

    def run(self, question, config="full"):
        if config not in CONFIGS:
            raise ValueError(f"config phải thuộc {tuple(CONFIGS)}, nhận {config!r}")
        if not question or not question.strip():
            raise ValueError("câu hỏi rỗng")
        if config not in self._graphs:
            self._graphs[config] = build_graph(self.deps, config)
        return to_response(self._graphs[config].invoke(initial_state(question.strip())))


def to_response(state):
    return {"answer": state["answer"], "refused": state["refused"], "citations": state["citations"],
            "retrieved_chunks": [{"chunk_id": c["chunk_id"], "score": c.get("score")} for c in state["evidence"]],
            "trace": state["trace"]}


def load_agent(variant="clause_header"):
    """Agent thật: LLM từ .env (LLM_*), Retriever trên biến thể đã nạp Chroma."""
    from src.generation.llm import from_env
    from src.retrieval.retriever import load_retriever
    laws = json.loads(LAWS_PATH.read_text(encoding="utf-8"))
    return Agent(Deps(from_env("LLM"), load_retriever(variant), laws))


def print_result(out):
    print(out["answer"])
    for c in out["citations"]:
        print(f"  [{c['chunk_id']}] {c['document']}, {c['article']}" + (f", {c['clause']}" if c["clause"] else ""))
    print("\nCác bước:")
    for step in out["trace"]:
        print(f"  {step['node']:<15}{step['ms']:>7} ms  {step['llm_calls']} LLM  "
              f"{json.dumps(step['output'], ensure_ascii=False)}")


def main(argv=None):
    parser = argparse.ArgumentParser(description="Hỏi tác tử pháp luật")
    parser.add_argument("question")
    parser.add_argument("--config", default="full", choices=list(CONFIGS))
    parser.add_argument("--variant", default="clause_header")
    parser.add_argument("--json", action="store_true", help="in nguyên phản hồi JSON (Phụ lục A.5)")
    args = parser.parse_args(argv)
    out = load_agent(args.variant).run(args.question, args.config)
    if args.json:

"""Chạy thử tác tử trên khoảng 10 câu thật (Sprint 3): in bảng tóm tắt, lưu đầu ra thô để đọc lỗi prompt.

python -m eval.agent_smoke [--config full]  -> eval/results/agent_smoke.jsonl, agent_smoke.md
"""
import argparse
import json
from pathlib import Path

from src.ingestion.legal_chunker import article_key, write_jsonl

CASES_PATH = Path("eval/test_cases.json")
OUT_DIR = Path("eval/results")
PICK = {"zalo": 3, "synthetic": 3, "manual": 4}
# dùng khi chưa có đủ câu manual của Hùng: phủ đủ nhánh direct, temporal, out_of_scope
EXTRA = [
    {"id": "X_direct", "source": "manual", "type": "direct", "question": "Điều 357 Bộ luật Dân sự 2015 quy định gì?",
     "gold_citations": ["BLDS2015_Dieu357"]},
    {"id": "X_temporal", "source": "manual", "type": "temporal",
     "question": "Năm 2022 tôi làm việc theo hợp đồng lao động xác định thời hạn 24 tháng, muốn nghỉ việc thì phải "
                 "báo trước bao nhiêu ngày?", "gold_citations": ["BLLD2019_Dieu35"]},
    {"id": "X_oos", "source": "manual", "type": "out_of_scope",
     "question": "Vượt đèn đỏ bằng xe máy bị phạt bao nhiêu tiền?", "gold_citations": []},
    {"id": "X_oos2", "source": "manual", "type": "out_of_scope",
     "question": "Hôm nay thời tiết ở Thành phố Hồ Chí Minh thế nào?", "gold_citations": []},
]


def pick_cases(cases):
    """3 zalo + 3 synthetic + tối đa 4 manual (ưu tiên câu không phải situational); thiếu manual thì bù bằng EXTRA."""
    chosen = []
    for source, n in PICK.items():
        pool = [c for c in cases if c["source"] == source]
        if source == "manual":
            pool = sorted(pool, key=lambda c: c["type"] == "situational")
        chosen = chosen + pool[:n]
    n_manual = sum(c["source"] == "manual" for c in chosen)
    return chosen + EXTRA[:PICK["manual"] - n_manual]


def summarize(case, out):
    cited = sorted({article_key(c["chunk_id"]) for c in out["citations"]})
    gold = sorted({article_key(g) for g in case["gold_citations"]})
    route = next((s["output"].get("route_type") for s in out["trace"] if s["node"] == "route"), "-")
    hit = bool(set(cited) & set(gold)) if gold else out["refused"]
    return {"id": case["id"], "type": case["type"], "route": route, "refused": out["refused"], "cited": cited,
            "gold": gold, "hit": hit, "llm_calls": sum(s["llm_calls"] for s in out["trace"]),
            "ms": sum(s["ms"] for s in out["trace"])}


def to_markdown(rows):
    lines = ["| id | type | route | refused | cited | gold | hit | LLM | ms |", "|" + "---|" * 9]
    lines += [f"| {r['id']} | {r['type']} | {r['route']} | {r['refused']} | {', '.join(r['cited'])} | "
              f"{', '.join(r['gold'])} | {r['hit']} | {r['llm_calls']} | {r['ms']} |" for r in rows]
    hits = sum(r["hit"] for r in rows)
    return "\n".join(lines + ["", f"Đúng ở cấp Điều (hoặc từ chối đúng): {hits}/{len(rows)}", "", "## Ghi chú lỗi prompt", ""])


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--config", default="full")
    args = parser.parse_args(argv)
    from src.agent.run import load_agent  # nạp mô hình chậm: chỉ khi chạy thật
    agent = load_agent()
    results = []
    for case in pick_cases(json.loads(CASES_PATH.read_text(encoding="utf-8"))):
        out = agent.run(case["question"], args.config)
        row = summarize(case, out)
        print(row, flush=True)
        results = results + [{"case": case, "response": out, "row": row}]
    write_jsonl(OUT_DIR / "agent_smoke.jsonl", results)
    (OUT_DIR / "agent_smoke.md").write_text(to_markdown([r["row"] for r in results]), encoding="utf-8")


if __name__ == "__main__":
    main()
```
