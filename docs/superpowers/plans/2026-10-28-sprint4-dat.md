ANY_DATE = "any"  # RQ4 "không lọc": lấy mọi phiên bản, kể cả văn bản đã hết hoặc chưa có hiệu lực


def check_date(value):
    """None, ANY_DATE hoặc ngày ISO đã chuẩn hóa; sai định dạng thì ValueError (ngày có thể do LLM trích ra)."""
    if value is None or value == ANY_DATE:
        return value
    return _date.fromisoformat(value).isoformat()


def is_valid_at(chunk, date=None):
    """date=None: chỉ văn bản còn hiệu lực; ANY_DATE: mọi văn bản; có date: effective_date <= date < expiry_date."""
    if date == ANY_DATE:
        return True
    if date is None:
        return chunk.get("expiry_date") is None
    date = check_date(date)
    expiry = chunk.get("expiry_date")
    return chunk["effective_date"] <= date and (expiry is None or date < expiry)


def chroma_where(date=None):
    """Điều kiện `where` của Chroma cho khoảng hiệu lực tại `date` (None: chỉ văn bản còn hiệu lực; ANY_DATE: không lọc, trả None), dùng trường số effective_from/effective_to."""
    if date == ANY_DATE:
        return None
    if date is None:
        return {"effective_to": {"$eq": OPEN_END}}
    d = int(check_date(date).replace("-", ""))
    return {"$and": [{"effective_from": {"$lte": d}}, {"effective_to": {"$gt": d}}]}
```

    def run(self, question, config="full", date_policy="event"):
        """Trả lời một câu hỏi.

        Args:
            question: câu hỏi pháp luật (tự bỏ khoảng trắng đầu cuối).
            config: một khóa của CONFIGS.
            date_policy: chính sách lọc hiệu lực (DATE_POLICIES); mặc định "event" = theo ngày của tình huống.

        Returns:
            dict theo Phụ lục A.5: answer, refused, citations, retrieved_chunks, trace.

        Raises:
            ValueError: config hoặc date_policy lạ, hoặc câu hỏi rỗng.
        """
        if config not in CONFIGS:
            raise ValueError(f"config phải thuộc {tuple(CONFIGS)}, nhận {config!r}")
        if date_policy not in DATE_POLICIES:
            raise ValueError(f"date_policy phải thuộc {DATE_POLICIES}, nhận {date_policy!r}")
        if not question or not question.strip():
            raise ValueError("câu hỏi rỗng")
        if config not in self._graphs:
            self._graphs[config] = build_graph(self.deps, config)
        return to_response(self._graphs[config].invoke(initial_state(question.strip(), date_policy)))
```

def citation_pr(cited, gold):
    """Precision và recall trích dẫn ở cấp Điều.

    Args:
        cited: khóa Điều được trích trong câu trả lời (có thể rỗng).
        gold: khóa Điều của căn cứ chuẩn (rỗng với câu out_of_scope).

    Returns:
        (precision, recall); precision = None khi không trích gì, recall = None khi gold rỗng.
    """
    cited, gold = set(cited), set(gold)
    hit = len(cited & gold)
    return (hit / len(cited) if cited else None, hit / len(gold) if gold else None)


def refusal_correct(case, response):
    """True khi tác tử từ chối đúng câu out_of_scope, hoặc không từ chối câu trong phạm vi."""
    return response["refused"] == (case["type"] == "out_of_scope")


def version_correct(case, cited_laws):
    """RQ4: câu temporal đúng phiên bản khi có trích dẫn và mọi văn bản được trích thuộc căn cứ chuẩn; câu khác trả None.

    Args:
        case: câu kiểm thử (cần type, gold_citations).
        cited_laws: law_id của các trích dẫn trong câu trả lời.
    """
    if case["type"] != "temporal":
        return None
    gold = {g.split("_", 1)[0] for g in case["gold_citations"]}
    return bool(cited_laws) and set(cited_laws) <= gold
```

    """Một dòng cho mỗi (config, source) và (config, 'all'); trung bình các cột `metrics`, bỏ qua giá trị None."""

"""Thực nghiệm sinh câu trả lời: RQ3 (5 cấu hình tác tử) và RQ4 (3 chính sách lọc hiệu lực).

Chạy dần qua nhiều ngày theo quota free tier: kết quả thô ghi nối vào JSONL, chạy lại thì bỏ qua câu đã xong.
  python -m eval.agent_eval run --rq rq3 --configs full baseline
  python -m eval.agent_eval run --rq rq3 --configs no_router no_grading no_verify --subset 60
  python -m eval.agent_eval run --rq rq4
  python -m eval.agent_eval report --rq rq3 [--subset 60]
"""
import argparse
import json
import random
import sys
from pathlib import Path

import openai

from eval.evaluate import _fmt, summarize
from eval.metrics import citation_pr, refusal_correct, version_correct
from src.ingestion.legal_chunker import article_key, read_jsonl

CASES_PATH = Path("eval/test_cases.json")
RESULTS_DIR = Path("eval/results")
SEED = 2026
RUNS = {
    # full trước: hết quota trong ngày thì vẫn đã có cấu hình chính
    "rq3": ["full", "baseline", "no_router", "no_grading", "no_verify"],
    "rq4": ["full@event", "full@current", "full@any"],
}
CASE_TYPES = {"rq3": None, "rq4": {"temporal"}}
METRICS = ("precision", "recall", "refusal_ok", "version_ok", "faithfulness", "correctness", "llm_calls", "ms")


def paths(rq):
    """Đường dẫn các file kết quả của một rq."""
    return {"results": RESULTS_DIR / f"agent_{rq}.jsonl", "judge": RESULTS_DIR / f"agent_{rq}_judge.jsonl",
            "report": RESULTS_DIR / f"agent_{rq}.md", "sheet": RESULTS_DIR / f"grading_{rq}.csv"}


def parse_run(run):
    """Tên run '<config>[@<chính sách>]' -> (config, chính sách); không có '@' thì chính sách 'event'."""
    config, _, policy = run.partition("@")
    return config, policy or "event"


def stratified(cases, n, seed=SEED):
    """Mẫu khoảng n câu giữ tỷ lệ theo `source`, giữ thứ tự gốc; cùng seed thì cùng mẫu."""
    rng = random.Random(seed)
    groups = {}
    for c in cases:
        groups.setdefault(c["source"], []).append(c)
    picked = set()
    for _, group in sorted(groups.items()):
        k = min(len(group), round(n * len(group) / len(cases)))
        picked |= {c["id"] for c in rng.sample(group, k)}
    return [c for c in cases if c["id"] in picked]


def select_cases(cases, rq, subset=None, limit=None):
    """Các câu dùng cho `rq`: rq4 chỉ câu temporal; `subset` = mẫu phân tầng; `limit` = N câu đầu (chạy thử)."""
    types = CASE_TYPES[rq]
    pool = [c for c in cases if types is None or c["type"] in types]
    if subset:
        pool = stratified(pool, subset)
    return pool[:limit] if limit else pool


def append_jsonl(path, row):
    """Ghi nối một dòng JSON (tạo thư mục cha nếu cần); ghi từng dòng để dừng giữa chừng không mất kết quả."""
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("a", encoding="utf-8") as f:
        f.write(json.dumps(row, ensure_ascii=False) + "\n")


def load_results(path):
    """{(id, run): response} của các dòng chạy thành công; dòng lỗi bị bỏ qua, trùng thì lấy dòng sau."""
    path = Path(path)
    rows = read_jsonl(path) if path.exists() else []
    return {(r["id"], r["run"]): r["response"] for r in rows if "error" not in r}


def run_agent(agent, cases, runs, out_path):
    """Chạy tác tử cho mọi (run, câu) chưa xong, ghi nối từng dòng vào `out_path`.

    Args:
        agent: có run(question, config, date_policy).
        cases: các câu cần chạy.
        runs: tên run, xem parse_run.
        out_path: file JSONL kết quả thô.

    Returns:
        Số dòng mới ghi. Hết quota (openai.RateLimitError sau khi SDK đã thử lại) thì dừng sớm, không ném lỗi;
        lỗi API khác ghi thành dòng {"error"} để lần chạy sau thử lại.
    """
    done = set(load_results(out_path))
    written = 0
    for run in runs:
        config, policy = parse_run(run)
        for case in cases:
            if (case["id"], run) in done:
                continue
            try:
                row = {"id": case["id"], "run": run, "response": agent.run(case["question"], config, policy)}
            except openai.RateLimitError:
                print(f"hết quota ở {run}/{case['id']}: dừng; chạy lại cùng lệnh khi quota hồi", file=sys.stderr)
                return written
            except openai.APIError as e:
                row = {"id": case["id"], "run": run, "error": repr(e)[:300]}
            append_jsonl(out_path, row)
            written += 1
    return written


def score(case, run, response):
    """Chỉ số của một câu trả lời ở cấp Điều; None = không áp dụng (vd recall của câu out_of_scope)."""
    cited = sorted({article_key(c["chunk_id"]) for c in response["citations"]})
    gold = sorted({article_key(g) for g in case["gold_citations"]})
    precision, recall = citation_pr(cited, gold)
    version = version_correct(case, [c.split("_", 1)[0] for c in cited])
    return {"id": case["id"], "config": run, "source": case["source"], "type": case["type"],
            "precision": precision, "recall": recall, "refusal_ok": float(refusal_correct(case, response)),
            "version_ok": None if version is None else float(version),
            "llm_calls": sum(s["llm_calls"] for s in response["trace"]), "ms": sum(s["ms"] for s in response["trace"])}


def report_rows(cases, results, judged):
    """Dòng chỉ số cho mọi kết quả thuộc `cases`, gộp điểm bộ chấm (`judged`: {(id, run): {...}}) nếu có."""
    by_id = {c["id"]: c for c in cases}
    rows = []
    for (cid, run), resp in results.items():
        if cid not in by_id:
            continue  # câu ngoài tập đang báo cáo (vd ngoài subset)
        verdict = judged.get((cid, run), {})
        rows.append({**score(by_id[cid], run, resp), "faithfulness": verdict.get("faithfulness"),
                     "correctness": verdict.get("correctness")})
    return rows


def to_markdown(summary):
    """Bảng Markdown: một dòng cho mỗi (run, source)."""
    head = ["| run | source | n | Citation P | Citation R | Từ chối đúng | Đúng phiên bản | Faithfulness | "
            "Correctness | LLM/câu | ms/câu |", "|" + "---|" * 11]
    lines = [f"| {s['config']} | {s['source']} | {s['n']} | " + " | ".join(_fmt(s[m]) for m in METRICS[:6])
             + f" | {_fmt(s['llm_calls'])} | {_fmt(s['ms'])} |" for s in summary]
    return "\n".join(head + lines) + "\n"


def _cases(args):
    return select_cases(json.loads(CASES_PATH.read_text(encoding="utf-8")), args.rq, args.subset, args.limit)


def cmd_run(args):
    from src.agent.run import load_agent  # nạp mô hình chậm: chỉ khi chạy thật
    agent = load_agent()
    runs = args.configs or RUNS[args.rq]
    written = run_agent(agent, _cases(args), runs, paths(args.rq)["results"])
    print(f"ghi {written} dòng; {agent.deps.llm.calls} lượt gọi API thật ({agent.deps.llm.requests} lượt logic)")


def cmd_report(args):
    p = paths(args.rq)
    judged = {(r["id"], r["run"]): r for r in read_jsonl(p["judge"])} if p["judge"].exists() else {}
    md = to_markdown(summarize(report_rows(_cases(args), load_results(p["results"]), judged), METRICS))
    p["report"].write_text(md, encoding="utf-8")
    print(md)


def build_parser():
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    sub = parser.add_subparsers(dest="command", required=True)
    for name, fn in (("run", cmd_run), ("report", cmd_report)):
        p = sub.add_parser(name)
        p.add_argument("--rq", choices=sorted(RUNS), required=True)
        p.add_argument("--subset", type=int, help="mẫu phân tầng khoảng N câu (cố định theo SEED)")
        p.add_argument("--limit", type=int, help="chỉ N câu đầu (chạy thử)")
        p.set_defaults(fn=fn)
        if name == "run":
            p.add_argument("--configs", nargs="+", help="tên run, vd full baseline full@any (mặc định: mọi run của rq)")
    return parser


def main(argv=None):
    """Điểm vào CLI."""
    args = build_parser().parse_args(argv)
    args.fn(args)


if __name__ == "__main__":
    main()

- Từ chối câu trong phạm vi: correctness = 0.

- Câu không có đáp án mẫu (Zalo): correctness = None.

        return self._wrap(self.replies[node].pop(0))

    @staticmethod
    def _wrap(reply):
        content = reply if isinstance(reply, str) else json.dumps(reply, ensure_ascii=False)
        return SimpleNamespace(choices=[SimpleNamespace(message=SimpleNamespace(content=content))])
```

def judge_messages(question, answer, contexts, reference):
    """Messages cho bộ chấm.

    Args:
        question: câu hỏi.
        answer: câu trả lời cần chấm.
        contexts: các chunk tác tử đã dùng làm ngữ cảnh.
        reference: đáp án mẫu; rỗng thì ghi rõ để bộ chấm trả correctness = null.
    """
    ref = f"Đáp án mẫu: {reference}" if reference else "Không có đáp án mẫu (correctness = null)."
    user = (f"Đoạn trích:\n{format_context(contexts)}\n\nCâu hỏi: {question}\n\n"
            f"Câu trả lời cần chấm:\n{answer}\n\n{ref}")
    return [{"role": "system", "content": f"{JUDGE_ROLE}\n\n{JUDGE_RULES}"}, {"role": "user", "content": user}]


def _score(value):
    try:
        value = float(value)
    except (TypeError, ValueError):
        return None
    return value if value in SCORES else None


def judge(llm, case, response, contexts):
    """Chấm một câu trả lời; tự chấm bằng code các trường hợp từ chối và câu ngoài phạm vi (không tốn lượt gọi).

    Args:
        llm: LLM bộ chấm (from_env("JUDGE")).
        case: câu kiểm thử (type, question, reference_answer).
        response: phản hồi Phụ lục A.5 của tác tử.
        contexts: {chunk_id: chunk} để lấy nội dung các đoạn trong `retrieved_chunks`.

    Returns:
        {"faithfulness", "correctness", "reason", "called"}; điểm thuộc {0, 0.5, 1} hoặc None (không áp dụng / lỗi).
    """
    if response["refused"]:
        return {"faithfulness": None, "correctness": float(case["type"] == "out_of_scope"), "reason": "từ chối",
                "called": False}
    if case["type"] == "out_of_scope":
        return {"faithfulness": None, "correctness": 0.0, "reason": "trả lời câu ngoài phạm vi", "called": False}
    reference = (case.get("reference_answer") or "").strip()
    used = [contexts[c["chunk_id"]] for c in response["retrieved_chunks"] if c["chunk_id"] in contexts]
    try:
        out = llm.chat_json(judge_messages(case["question"], response["answer"], used, reference))
    except ValueError:
        return {"faithfulness": None, "correctness": None, "reason": "bộ chấm không trả JSON hợp lệ", "called": True}
    return {"faithfulness": _score(out.get("faithfulness")),
            "correctness": _score(out.get("correctness")) if reference else None,
            "reason": str(out.get("reason") or "")[:300], "called": True}


def load_contexts():
    """{chunk_id: chunk} từ file chunk Khoản rồi file chunk Điều (câu direct dùng chunk Điều làm ngữ cảnh)."""
    # shortcut: Điều có dẫn nhập trùng chunk_id với chunk Điều nên bản Điều (đầy đủ hơn) đè lên; ngữ cảnh chấm rộng hơn
    # một chút so với ngữ cảnh tác tử thấy. Lưu nội dung bằng chứng trong response nếu cần chấm chặt hơn.
    return {c["chunk_id"]: c for path in (OUT_PATH, ARTICLE_OUT_PATH) for c in read_jsonl(path)}

def judge_results(llm, cases, results, out_path, contexts):
    """Chấm mọi kết quả thuộc `cases` chưa có trong `out_path`, ghi nối từng dòng; hết quota thì dừng sớm.

    Returns:
        Số dòng mới ghi.
    """
    from eval.judge import judge
    out_path = Path(out_path)
    done = {(r["id"], r["run"]) for r in read_jsonl(out_path)} if out_path.exists() else set()
    by_id = {c["id"]: c for c in cases}
    written = 0
    for (cid, run), resp in results.items():
        if (cid, run) in done or cid not in by_id:
            continue
        try:
            verdict = judge(llm, by_id[cid], resp, contexts)
        except openai.RateLimitError:
            print(f"bộ chấm hết quota ở {run}/{cid}: dừng; chạy lại khi quota hồi", file=sys.stderr)
            return written
        except openai.APIError as e:
            print(f"bỏ qua {run}/{cid}: {e!r}"[:300], file=sys.stderr)
            continue
        append_jsonl(out_path, {"id": cid, "run": run, **verdict})
        written += 1
    return written


SHEET_FIELDS = ("id", "question", "answer", "contexts", "reference_answer", "faithfulness", "correctness", "note")


def write_sheet(cases, results, run, n, path, contexts):
    """Phiếu chấm tay CSV (UTF-8 BOM để Excel đọc đúng tiếng Việt): mẫu cố định n câu đã được trả lời của `run`.

    Không kèm điểm của bộ chấm LLM, để người chấm không bị ảnh hưởng. Trả số câu đã ghi.
    """
    from src.generation.prompts import format_context
    answered = [c for c in cases if (c["id"], run) in results and not results[(c["id"], run)]["refused"]]
    rows = []
    for c in stratified(answered, n) if len(answered) > n else answered:
        resp = results[(c["id"], run)]
        used = [contexts[x["chunk_id"]] for x in resp["retrieved_chunks"] if x["chunk_id"] in contexts]
        rows.append({"id": c["id"], "question": c["question"], "answer": resp["answer"],
                     "contexts": format_context(used), "reference_answer": c.get("reference_answer", ""),
                     "faithfulness": "", "correctness": "", "note": ""})
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", newline="", encoding="utf-8-sig") as f:
        writer = csv.DictWriter(f, fieldnames=SHEET_FIELDS)
        writer.writeheader()
        writer.writerows(rows)
    return len(rows)


def _human_score(text):
    text = (text or "").strip().replace(",", ".")  # người chấm có thể gõ 0,5
    return float(text) if text else None


def agreement(sheet_rows, judged, run):
    """Tỷ lệ khớp điểm giữa người chấm (phiếu đã điền) và bộ chấm LLM, theo từng tiêu chí.

    Returns:
        {tiêu chí: (tỷ lệ khớp hoặc None, số cặp được so)}; ô trống hoặc điểm LLM None thì không tính.
    """
    out = {}
    for metric in ("faithfulness", "correctness"):
        pairs = [(_human_score(r[metric]), judged[(r["id"], run)][metric]) for r in sheet_rows
                 if _human_score(r[metric]) is not None and judged.get((r["id"], run), {}).get(metric) is not None]
        out[metric] = (sum(a == b for a, b in pairs) / len(pairs) if pairs else None, len(pairs))
    return out

"""API hỏi đáp pháp luật (Phụ lục A.5): POST /api/ask, GET /api/article/{law_id}/{article}, GET /health.

Chạy: uvicorn src.api.main:app --port 8000   (nạp mô hình một lần khi khởi động, ~30 s)
"""
import logging
import threading
from contextlib import asynccontextmanager
from typing import Literal

import openai
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, ConfigDict, Field

from src.agent.state import CONFIGS

log = logging.getLogger("vn_legal_qa.api")
MAX_QUESTION = 1000
LLM_ERROR = "Dịch vụ mô hình ngôn ngữ đang quá tải hoặc tạm thời không khả dụng. Vui lòng thử lại sau ít phút."


class AskRequest(BaseModel):
    """Thân request của POST /api/ask; khoảng trắng đầu cuối bị bỏ trước khi kiểm tra độ dài."""

    model_config = ConfigDict(str_strip_whitespace=True)
    question: str = Field(min_length=1, max_length=MAX_QUESTION)
    config: Literal[tuple(CONFIGS)] = "full"


def _load_real_agent():
    from src.agent.run import load_agent
    return load_agent()


def create_app(load=_load_real_agent):
    """Tạo app FastAPI.

    Args:
        load: hàm không tham số trả Agent; gọi một lần khi app khởi động (test truyền tác tử giả).
    """
    state = {}
    # LLM (bộ đếm lượt gọi trong trace, giãn cách RPM, cache) không an toàn đa luồng: phục vụ tuần tự từng câu hỏi
    lock = threading.Lock()

    @asynccontextmanager
    async def lifespan(app):
        state["agent"] = load()
        yield

    app = FastAPI(title="Hỏi đáp pháp luật Việt Nam", lifespan=lifespan)

    @app.post("/api/ask")
    def ask(req: AskRequest):
        """Trả lời một câu hỏi theo Phụ lục A.5; LLM lỗi sau khi đã thử lại thì 503."""
        try:
            with lock:
                return state["agent"].run(req.question, req.config)
        except openai.APIError:
            log.exception("LLM lỗi khi trả lời câu hỏi: %.100s", req.question)
            raise HTTPException(status_code=503, detail=LLM_ERROR)

    @app.get("/api/article/{law_id}/{article}")
    def article(law_id: str, article: str):
        """Toàn văn một Điều (chunk cấp Điều); 404 nếu không có."""
        found = state["agent"].deps.retriever.get_article(law_id, article)
        if found is None:
            raise HTTPException(status_code=404, detail=f"Không có {law_id} Điều {article}")
        return found

    @app.get("/health")
    def health():
        return {"status": "ok", "model": state["agent"].deps.llm.model}

    return app


app = create_app()
