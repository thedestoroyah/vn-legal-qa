```python
# tests/test_legal_chunker.py
def test_read_jsonl_skips_truncated_last_line(tmp_path, capsys):
    p = tmp_path / "r.jsonl"
    p.write_text('{"a": 1}\n{"a": 2}\n{"a": ', encoding="utf-8")
    assert read_jsonl(p) == [{"a": 1}, {"a": 2}]
    assert "r.jsonl" in capsys.readouterr().err

def test_read_jsonl_still_rejects_broken_line_in_middle(tmp_path):
    p = tmp_path / "r.jsonl"
    p.write_text('{"a": 1}\n{"a": \n{"a": 3}\n', encoding="utf-8")
    with pytest.raises(json.JSONDecodeError):
        read_jsonl(p)

# tests/test_retriever.py  (chunk, Retriever, FakeIndex có sẵn trong file)
def test_get_article_ignores_case():
    r = Retriever([chunk("BLDS2015_Dieu50a_Khoan1", "A", "Khoản 1")], FakeIndex([]), FakeIndex([]))
    assert r.get_article("blds2015", "50A")["chunk_id"] == "BLDS2015_Dieu50a"

# tests/test_api.py
def test_index_error_returns_json_500():
    class Broken:
        def run(self, *a, **k): raise RuntimeError("chỉ mục lệch")
    r = TestClient(create_app(lambda: Broken())).post("/api/ask", json={"question": "q"})
    assert r.status_code == 500 and r.json()["detail"] == INDEX_ERROR

# tests/test_agent_eval.py
def test_judge_results_does_not_record_broken_json_verdict(tmp_path, monkeypatch):
    monkeypatch.setattr("eval.judge.judge", lambda llm, case, resp, ctx:
                        {"faithfulness": None, "correctness": None, "reason": "bộ chấm không trả JSON hợp lệ", "called": True})
    path = tmp_path / "j.jsonl"
    assert agent_eval.judge_results(None, [case("A")], {("A", "full"): response()}, path, {}) == 0
    assert not path.exists()
```
